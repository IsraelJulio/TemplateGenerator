== Fonte integral de mut.sh
#!/usr/bin/env bash
# Uso: mut.sh <id> <arquivo> <sed-expr> <args do playwright...>
# Aplica a mutacao, mostra o diff, roda o Playwright, restaura e confere a restauracao.
set -u
S=/c/Users/2273129/AppData/Local/Temp/claude/c--Users-2273129-Documents-Projects-TemplateGenerator/318d2cf1-dde3-4bfe-96a2-838beef3a510/scratchpad/t09b
ID=$1; FILE=$2; EXPR=$3; shift 3
cd /c/Users/2273129/Documents/Projects/TemplateGenerator/src/web
export PATH="/c/Users/2273129/AppData/Local/Microsoft/dotnet:$PATH" DOTNET_ROOT='C:\Users\2273129\AppData\Local\Microsoft\dotnet'
OUT=$S/$ID.log
cp "$FILE" "$S/$ID.bak"
{
  echo "\$ cp $FILE $ID.bak"
  echo "\$ sed -i \"$EXPR\" $FILE"
  sed -i "$EXPR" "$FILE"
  echo "\$ diff $ID.bak $FILE"
  diff "$S/$ID.bak" "$FILE"
  echo "diff exit=$?   (1 = a mutacao foi aplicada)"
  echo "\$ npx playwright test$(printf " %q" "$@")"
  echo "(linhas \"[WebServer]\" removidas com: grep -v '^\[WebServer\]')"
  echo "inicio: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  npx playwright test "$@" > "$S/$ID.raw.txt" 2>&1
  RC=$?
  grep -v '^\[WebServer\]' "$S/$ID.raw.txt"
  echo "exit=$RC"
  echo "\$ cp $ID.bak $FILE && diff $ID.bak $FILE && echo restaurado"
  cp "$S/$ID.bak" "$FILE" && diff "$S/$ID.bak" "$FILE" && echo restaurado
} > "$OUT" 2>&1
tail -5 "$OUT"
