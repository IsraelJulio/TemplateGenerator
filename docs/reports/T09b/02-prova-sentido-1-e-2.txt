const fs = require('fs');
const report = JSON.parse(fs.readFileSync(process.argv[2], 'utf8'));
const strip = (s) => s.replace(/\u001b\[[0-9;]*m/g, '');
const walk = (suite, out) => {
  for (const spec of suite.specs ?? []) {
    for (const t of spec.tests) {
      for (const r of t.results) {
        out.push({ file: spec.file, title: spec.title, project: t.projectName, expectedStatus: t.expectedStatus, status: r.status, outcome: t.status, error: r.errors?.[0]?.message });
      }
    }
  }
  for (const s of suite.suites ?? []) walk(s, out);
  return out;
};
const rows = report.suites.flatMap((s) => walk(s, []));
for (const row of rows) {
  console.log(`--- [${row.project}] ${row.file} > ${row.title}`);
  console.log(`    expectedStatus=${row.expectedStatus}  status=${row.status}  outcome=${row.outcome}`);
  if (row.error) console.log(strip(row.error).split('\n').map((l) => '    | ' + l).join('\n'));
}
console.log(`stats: ${JSON.stringify(report.stats)}`);
